// RegisterNodeRequest for POST /api/v1/nodes
type RegisterNodeRequest struct {
    GPUUUID     string `json:"gpu_uuid"`
    GPUType     string `json:"gpu_type"`
    MemoryGB    int    `json:"memory_gb"`
    PricePerSec string `json:"price_per_sec"`
}

// RegisterNode registers a GPU node for the provider
// POST /api/v1/nodes (requires auth)
func (c *HubClient) RegisterNode(ctx context.Context, req RegisterNodeRequest) (map[string]interface{}, error) {
    jsonBody, err := json.Marshal(req)
    if err != nil {
        return nil, fmt.Errorf("failed to marshal request: %w", err)
    }

    httpReq, err := http.NewRequestWithContext(ctx, http.MethodPost, c.baseURL+"/api/v1/nodes", bytes.NewReader(jsonBody))
    if err != nil {
        return nil, fmt.Errorf("failed to create request: %w", err)
    }
    httpReq.Header.Set("Content-Type", "application/json")
    if c.authToken != "" {
        httpReq.Header.Set("Authorization", "Bearer "+c.authToken)
    }

    resp, err := c.httpClient.Do(httpReq)
    if err != nil {
        return nil, fmt.Errorf("request failed: %w", err)
    }
    defer resp.Body.Close()

    if resp.StatusCode != http.StatusCreated && resp.StatusCode != http.StatusOK {
        body, _ := io.ReadAll(resp.Body)
        return nil, fmt.Errorf("unexpected status %d: %s", resp.StatusCode, string(body))
    }

    var result map[string]interface{}
    if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
        return nil, fmt.Errorf("failed to decode response: %w", err)
    }

    return result, nil
}

// TestTwoUserRentalFlowE2E tests the complete rental lifecycle with two users:
// - Provider: registers node, gets certificate, receives rental notification
// - Renter: deposits tokens, creates session, confirms with blockchain, gets SSH info
//
// This test requires:
// - Running docker-compose.e2e.yml (Hub, Hardhat, PostgreSQL, worldland-node)
// - AUTH_DISABLED=true on Hub for simplicity (or implement SIWE auth)
func TestTwoUserRentalFlowE2E(t *testing.T) {
    cfg := DefaultConfig()

    // Create separate clients for Provider and Renter
    providerClient := NewHubClient(cfg.HubURL)
    renterClient := NewHubClient(cfg.HubURL)

    // Hardhat accounts
    renterAddress := HardhatAccount1Address      // Account #1
    renterPrivateKey := HardhatAccount1PrivateKey
    providerAddress := HardhatAccount2Address    // Account #2
    // providerPrivateKey := HardhatAccount2PrivateKey

    // Hardhat node URL
    hardhatURL := getEnvOrDefault("HARDHAT_URL", "http://localhost:8545")
    blockchainHelper := NewBlockchainHelper(hardhatURL)

    var contractHelper *ContractHelper
    var providerNodeID string
    var sessionID string
    var startRentalTxHash string

    feature := features.New("Two-User Rental Flow E2E").
        Setup(func(ctx context.Context, t *testing.T, ecfg *envconf.Config) context.Context {
            // Check Hub availability
            if !isHubAvailable(ctx, renterClient) {
                t.Skip("Hub server not available at " + cfg.HubURL)
            }

            // Check Hardhat availability
            if !blockchainHelper.client.IsHealthy(ctx) {
                t.Skip("Hardhat node not available at " + hardhatURL)
            }

            // Initialize contract helper
            var err error
            contractHelper, err = NewContractHelperFromDeployment(blockchainHelper)
            if err != nil {
                t.Skipf("Failed to load contract addresses: %v", err)
            }

            // Create EVM snapshot
            if err := blockchainHelper.Snapshot(ctx); err != nil {
                t.Logf("WARNING: Failed to create EVM snapshot: %v", err)
            }

            return ctx
        }).
        // ==================== PROVIDER FLOW ====================
        Assess("Provider: Register GPU node", func(ctx context.Context, t *testing.T, ecfg *envconf.Config) context.Context {
            // Register a GPU node as Provider
            req := RegisterNodeRequest{
                GPUUUID:     "GPU-E2E-TEST-001",
                GPUType:     "NVIDIA RTX 4090",
                MemoryGB:    24,
                PricePerSec: "1000000", // 1000000 wei per second
            }

            result, err := providerClient.RegisterNode(ctx, req)
            if err != nil {
                t.Fatalf("Failed to register node: %v", err)
            }

            nodeID, ok := result["node_id"].(string)
            if !ok {
                t.Fatalf("Node registration response missing node_id: %v", result)
            }

            providerNodeID = nodeID
            t.Logf("Provider: Registered GPU node: %s", providerNodeID)

            return ctx
        }).
        Assess("Provider: Get mTLS certificate", func(ctx context.Context, t *testing.T, ecfg *envconf.Config) context.Context {
            if providerNodeID == "" {
                t.Skip("No node ID from previous step")
            }

            certResult, err := providerClient.GetNodeCertificate(ctx, providerNodeID)
            if err != nil {
                t.Fatalf("Failed to get node certificate: %v", err)
            }

            // Verify certificate fields
            if _, ok := certResult["certificate"].(string); !ok {
                t.Error("Certificate response missing 'certificate' field")
            }
            if _, ok := certResult["private_key"].(string); !ok {
                t.Error("Certificate response missing 'private_key' field")
            }
            if expiresAt, ok := certResult["expires_at"].(string); ok {
                t.Logf("Provider: Certificate issued, expires: %s", expiresAt)
            }

            return ctx
        }).
        Assess("Provider: Verify node is listed", func(ctx context.Context, t *testing.T, ecfg *envconf.Config) context.Context {
            nodes, err := providerClient.ListNodes(ctx)
            if err != nil {
                t.Fatalf("Failed to list nodes: %v", err)
            }

            found := false
            for _, node := range nodes {
                if id, _ := node["id"].(string); id == providerNodeID {
                    found = true
                    t.Logf("Provider: Node %s found in list with status: %v", providerNodeID, node["status"])
                    break
                }
            }

            if !found {
                t.Errorf("Provider: Node %s not found in provider's node list", providerNodeID)
            }

            return ctx
        }).
        // ==================== RENTER FLOW ====================
        Assess("Renter: Setup tokens (mint, approve, deposit)", func(ctx context.Context, t *testing.T, ecfg *envconf.Config) context.Context {
            // Mint 1000 tokens to Renter
            tokenAmount := new(big.Int)
            tokenAmount.SetString("1000000000000000000000", 10) // 1000 * 10^18

            t.Log("Renter: Minting tokens...")
            _, err := contractHelper.MintTokens(ctx, renterAddress, tokenAmount)
            if err != nil {
                t.Fatalf("Failed to mint tokens: %v", err)
            }

            t.Log("Renter: Approving rental contract...")
            _, err = contractHelper.ApproveToken(ctx, renterPrivateKey, tokenAmount)
            if err != nil {
                t.Fatalf("Failed to approve tokens: %v", err)
            }

            t.Log("Renter: Depositing tokens to rental contract...")
            _, err = contractHelper.DepositTokens(ctx, renterPrivateKey, tokenAmount)
            if err != nil {
                t.Fatalf("Failed to deposit tokens: %v", err)
            }

            // Verify deposit
            deposit, err := contractHelper.GetContractDeposit(ctx, renterAddress)
            if err != nil {
                t.Fatalf("Failed to get deposit balance: %v", err)
            }
            t.Logf("Renter: Deposit balance: %s", deposit.String())

            return ctx
        }).
        Assess("Renter: Create rental session", func(ctx context.Context, t *testing.T, ecfg *envconf.Config) context.Context {
            if providerNodeID == "" {
                t.Skip("No provider node ID from provider flow")
            }

            // Create session targeting the provider's node
            session, err := renterClient.CreateSession(ctx, providerNodeID, "1000000", "busybox:latest")
            if err != nil {
                t.Fatalf("Failed to create session: %v", err)
            }

            // Extract session ID
            id, ok := session["id"].(string)
            if !ok {
                id, ok = session["sessionId"].(string)
            }
            if !ok {
                t.Fatalf("Session response missing 'id' or 'sessionId': %v", session)
            }

            sessionID = id
            t.Logf("Renter: Created rental session: %s", sessionID)

            // Verify initial state
            if state, _ := session["state"].(string); state != "PENDING" {
                t.Logf("WARNING: Expected PENDING state, got: %s", state)
            }

            return ctx
        }).
        Assess("Renter: Call startRental on blockchain", func(ctx context.Context, t *testing.T, ecfg *envconf.Config) context.Context {
            if sessionID == "" {
                t.Skip("No session ID from previous step")
            }

            pricePerSecond := big.NewInt(1000000)

            t.Logf("Renter: Calling startRental (provider: %s)", providerAddress)
            txHash, err := contractHelper.StartRental(ctx, renterPrivateKey, providerAddress, pricePerSecond)
            if err != nil {
                t.Fatalf("Failed to call startRental: %v", err)
            }

            startRentalTxHash = txHash
            t.Logf("Renter: startRental tx: %s", txHash)

            return ctx
        }).
        Assess("Renter: Confirm session with txHash", func(ctx context.Context, t *testing.T, ecfg *envconf.Config) context.Context {
            if sessionID == "" || startRentalTxHash == "" {
                t.Skip("Missing session ID or txHash")
            }

            result, err := renterClient.ConfirmSession(ctx, sessionID, startRentalTxHash)
            if err != nil {
                t.Fatalf("Failed to confirm session: %v", err)
            }

            t.Logf("Renter: Session confirmation submitted")
            if state, ok := result["state"].(string); ok {
                t.Logf("  State after confirm: %s", state)
            }

            return ctx
        }).
        Assess("Wait for session to transition to RUNNING", func(ctx context.Context, t *testing.T, ecfg *envconf.Config) context.Context {
            if sessionID == "" {
                t.Skip("No session ID")
            }

            // Wait for ConfirmationWorker to verify and transition session
            deadline := time.Now().Add(60 * time.Second)
            var lastState string

            for time.Now().Before(deadline) {
                session, err := renterClient.GetSession(ctx, sessionID)
                if err != nil {
                    t.Logf("Error getting session: %v", err)
                    time.Sleep(5 * time.Second)
                    continue
                }

                state, _ := session["state"].(string)
                lastState = state

                if state == "RUNNING" {
                    t.Logf("Session transitioned to RUNNING")

                    // Check SSH info
                    if sshInfo, ok := session["sshConnectionInfo"].(map[string]interface{}); ok {
                        t.Logf("  SSH Host: %v", sshInfo["host"])
                        t.Logf("  SSH Port: %v", sshInfo["port"])
                    }
                    return ctx
                }

                if state == "FAILED" {
                    reason, _ := session["failureReason"].(string)
                    t.Fatalf("Session FAILED: %s", reason)
                }

                t.Logf("  State: %s (waiting for RUNNING...)", state)
                time.Sleep(5 * time.Second)
            }

            t.Errorf("Session did not transition to RUNNING within 60s. Last state: %s", lastState)
            return ctx
        }).
        Assess("Verify worldland-node received rental notification", func(ctx context.Context, t *testing.T, ecfg *envconf.Config) context.Context {
            // In a full E2E with worldland-node running, we would:
            // 1. Check Node's /rentals/status?sessionId=xxx endpoint
            // 2. Verify container was created
            //
            // For now, if session reached RUNNING and has SSH info, Node was notified
            if sessionID == "" {
                t.Skip("No session ID")
            }

            session, err := renterClient.GetSession(ctx, sessionID)
            if err != nil {
                t.Logf("Warning: Could not verify node notification: %v", err)
                return ctx
            }

            // If session has SSH info from Node, notification succeeded
            if sshInfo, ok := session["sshConnectionInfo"].(map[string]interface{}); ok {
                if host, _ := sshInfo["host"].(string); host != "" {
                    t.Logf("worldland-node received notification (SSH host: %s)", host)
                }
            }

            return ctx
        }).
        Assess("Renter: Terminate session", func(ctx context.Context, t *testing.T, ecfg *envconf.Config) context.Context {
            if sessionID == "" {
                t.Skip("No session ID")
            }

            result, err := renterClient.TerminateSession(ctx, sessionID)
            if err != nil {
                t.Fatalf("Failed to terminate session: %v", err)
            }

            t.Logf("Renter: Session terminated: %v", result)

            // Wait for state change
            deadline := time.Now().Add(30 * time.Second)
            for time.Now().Before(deadline) {
                session, _ := renterClient.GetSession(ctx, sessionID)
                if state, _ := session["state"].(string); state == "STOPPED" || state == "TERMINATED" {
                    t.Logf("Renter: Session final state: %s", state)
                    return ctx
                }
                time.Sleep(2 * time.Second)
            }

            return ctx
        }).
        Teardown(func(ctx context.Context, t *testing.T, ecfg *envconf.Config) context.Context {
            // Revert EVM snapshot
            if blockchainHelper.hasSnapshot {
                if err := blockchainHelper.Revert(ctx); err != nil {
                    t.Logf("Warning: Failed to revert snapshot: %v", err)
                }
            }
            return ctx
        }).
