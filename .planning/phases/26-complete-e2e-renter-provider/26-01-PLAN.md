Create a certificate generation script for E2E tests:

1. Create `/home/ahwlsqja/worldland-backend/e2e/scripts/generate-certs.sh`:
```bash
#!/bin/bash
set -e

CERT_DIR="${1:-$(dirname $0)/../certs}"
mkdir -p "$CERT_DIR"

# Generate CA
openssl genrsa -out "$CERT_DIR/ca.key" 4096
openssl req -x509 -new -nodes -key "$CERT_DIR/ca.key" -sha256 -days 365 \
    -out "$CERT_DIR/ca.crt" -subj "/CN=E2E Test CA"

# Generate Hub server certificate
openssl genrsa -out "$CERT_DIR/hub.key" 2048
openssl req -new -key "$CERT_DIR/hub.key" -out "$CERT_DIR/hub.csr" \
    -subj "/CN=hub" \
    -addext "subjectAltName=DNS:hub,DNS:localhost,IP:127.0.0.1"
openssl x509 -req -in "$CERT_DIR/hub.csr" -CA "$CERT_DIR/ca.crt" -CAkey "$CERT_DIR/ca.key" \
    -CAcreateserial -out "$CERT_DIR/hub.crt" -days 365 -sha256 \
    -extfile <(printf "subjectAltName=DNS:hub,DNS:localhost,IP:127.0.0.1")

# Generate Hub client certificate (for calling Node)
openssl genrsa -out "$CERT_DIR/hub-client.key" 2048
openssl req -new -key "$CERT_DIR/hub-client.key" -out "$CERT_DIR/hub-client.csr" \
    -subj "/CN=hub-client"
openssl x509 -req -in "$CERT_DIR/hub-client.csr" -CA "$CERT_DIR/ca.crt" -CAkey "$CERT_DIR/ca.key" \
    -CAcreateserial -out "$CERT_DIR/hub-client.crt" -days 365 -sha256

# Generate Node certificate
openssl genrsa -out "$CERT_DIR/node.key" 2048
openssl req -new -key "$CERT_DIR/node.key" -out "$CERT_DIR/node.csr" \
    -subj "/CN=worldland-node" \
    -addext "subjectAltName=DNS:worldland-node,DNS:localhost,IP:127.0.0.1"
openssl x509 -req -in "$CERT_DIR/node.csr" -CA "$CERT_DIR/ca.crt" -CAkey "$CERT_DIR/ca.key" \
    -CAcreateserial -out "$CERT_DIR/node.crt" -days 365 -sha256 \
    -extfile <(printf "subjectAltName=DNS:worldland-node,DNS:localhost,IP:127.0.0.1")

# Cleanup CSRs
rm -f "$CERT_DIR"/*.csr

echo "Certificates generated in $CERT_DIR"
ls -la "$CERT_DIR"
```

3. Add to .gitignore: `e2e/certs/*.key` (keep certs but not private keys in repo)
